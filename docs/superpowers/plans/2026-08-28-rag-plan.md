```ts
export function stripSignatures(b: string) {
  const m = b.split('\n--\n')[0];
  return (m ?? b).split('\nEnviado desde')[0].trim();
}
export function chunkText(t: string, { size, overlap }: { size: number; overlap: number }) {
  const r = [];
  for (let i = 0; i < t.length; i += size - overlap) r.push(t.slice(i, i + size));
  return r.filter(Boolean);
}
```
